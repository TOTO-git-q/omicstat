# requires: survival ggplot2 jsonlite digest
here <- dirname(normalizePath(sub("^--file=", "", grep("^--file=", commandArgs(FALSE), value = TRUE)[1])))
mod <- dirname(here); shared <- file.path(mod, "..", "_shared", "scripts")
for (f in c("module_kit.R", "handoff.R", "survival_io.R", "fig_all.R", "calls_check.R")) source(file.path(shared, f))
source(file.path(mod, "scripts/survival_expr_core.R")); source(file.path(mod, "scripts/survival_expr_report.R"))
check <- function(ok, msg) { if (!isTRUE(ok)) stop(paste("FAIL:", msg)); cat("PASS:", msg, "\n") }
refuse <- function(expr, code) { z <- tryCatch({ force(expr); "none" }, error = conditionMessage); check(grepl(code, z, fixed = TRUE), paste("typed refusal", code, z)) }
tmp <- tempfile("survival-smoke-"); dir.create(tmp)
ret <- system2("python3", c(shQuote(file.path(here, "prepare_acceptance.py")), shQuote(file.path(tmp,"fixtures"))))
check(ret == 0, "real limma-array upstream producer")
fixture <- file.path(tmp, "fixtures"); v <- jsonlite::fromJSON(file.path(here,"acceptance.json"))$canonical$argv
v <- gsub("{fixtures}", fixture, v, fixed = TRUE); a <- list(); i <- 1L
while (i <= length(v)) { k <- sub("^--", "", v[i]); if (k == "development-mode") {a[[k]] <- TRUE; i <- i + 1L} else {a[[k]] <- v[i+1L]; i <- i+2L} }
for (k in c("min-events","max-iter","cox-eps","ph-alpha","family-size","ci-level","seed")) a[[k]] <- as.numeric(a[[k]])
a$outdir <- file.path(tmp,"render"); dir.create(a$outdir)
x <- survival_expr_compute(a)
cd <- read.csv(a$clinical); ex <- read.csv(file.path(fixture,"expression.csv"),check.names=FALSE); cd$signal <- as.numeric(ex[1,match(cd$unit,names(ex))])
check(x$metadata$n_biological == 160 && x$metadata$n_events == sum(cd$event), "synthetic units and explicit 0/1 events")
reference <- survival::coxph(survival::Surv(time,event)~signal+covar1+covar2,data=cd,ties="efron",control=survival::coxph.control(iter.max=50,eps=1e-9))
check(abs(x$tables$effect_table$coefficient - coef(reference)[1]) < 1e-10, "direct native Cox parity on project-generated fixture")
check(all(x$tables$km_risk$n_risk[x$tables$km_risk$time==0] == table(x$tables$km_membership$group)), "risk counts match subjects at zero")
check(nrow(x$tables$ph_tests)==4 && x$tables$ph_tests$term[4]=="GLOBAL", "PH diagnostics include covariates and global")
b <- a; b$scaling <- "zscore"; y <- survival_expr_compute(b)
clinical_sd <- sd(as.numeric(read.csv(file.path(fixture,"expression.csv"),check.names=FALSE)[1,-1]))
check(abs(y$tables$effect_table$coefficient - x$tables$effect_table$coefficient * clinical_sd) < 1e-9, "per-SD effect rescales coefficient without changing hypothesis")
check(abs(y$tables$effect_table$p_value - x$tables$effect_table$p_value) < 1e-10, "scaling preserves continuous Cox test")
b <- a; b$cutoff <- "prespecified"; b$cutpoint <- 0; y <- survival_expr_compute(b)
check(all(y$tables$km_membership$group == ifelse(y$tables$km_membership$expression <= 0,"low","high")), "prespecified cutoff on supplied expression scale")
b <- a; b$cutoff <- "optimal"; b$exploratory <- TRUE; b[["min-group-fraction"]] <- .2; y <- survival_expr_compute(b)
check(is.na(y$metadata$logrank_p) && is.na(y$tables$km_test$p_value) && nrow(y$tables$cutpoint_search)>1,"optimal search exports all candidates and withholds confirmatory p")
check(y$tables$km_test$chisq==max(y$tables$cutpoint_search$chisq),"explicit deterministic maximum logrank search")
survival_expr_render(y,a$outdir,"survival-expr")
sp <- jsonlite::fromJSON(file.path(a$outdir,"figures/km.figure_spec.json"))
check(grepl("exploratory",sp$caption,ignore.case=TRUE),"exploratory warning travels in figure spec")
b <- a; b$genes <- "SIM_SIGNAL,SIM2"; b[["family-size"]] <- 3; b$adjust <- "holm"; y <- survival_expr_compute(b)
check(nrow(y$tables$effect_table)==2 && isTRUE(all.equal(y$tables$effect_table$p_adjusted,p.adjust(y$tables$effect_table$p_value,"holm",n=3))),"planned multi-gene family adjustment")
b <- a; b$covariates <- "none"; y <- survival_expr_compute(b)
check(nrow(y$tables$coefficients)==1,"no-covariate continuous model supported explicitly")
b <- a; b$ties <- "breslow"; check(is.finite(survival_expr_compute(b)$tables$effect_table$estimate),"Breslow ties executes")
b <- a; b$ties <- "exact"; check(is.finite(survival_expr_compute(b)$tables$effect_table$estimate),"exact ties executes")
# Clinical row order does not change expression/subject alignment.
c <- read.csv(a$clinical,colClasses="character",check.names=FALSE); write.csv(c[nrow(c):1,],file.path(tmp,"reverse.csv"),row.names=FALSE)
b <- a; b$clinical <- file.path(tmp,"reverse.csv"); y <- survival_expr_compute(b)
check(identical(x$tables$effect_table,y$tables$effect_table),"identity-based row permutation invariance")
for (case in c("batch","display","candidate","wrong-type","tampered")) { b <- a; b[["from-run"]] <- file.path(fixture,case); refuse(survival_expr_compute(b),switch(case,batch="OS_BATCH_CORRECTED_INPUT",display="OS_DISPLAY_ONLY_INPUT","OS_HANDOFF_MISMATCH")) }
b <- a; b$clinical <- file.path(fixture,"missing-subject.csv"); refuse(survival_expr_compute(b),"OS_INVALID_DESIGN")
b <- a; b$clinical <- file.path(fixture,"missing-covariate.csv"); refuse(survival_expr_compute(b),"OS_INVALID_INPUT")
b <- a; b$cutoff <- "optimal"; refuse(survival_expr_compute(b),"OS_MISSING_PARAMETER")
b <- a; b$cutoff <- "prespecified"; b$cutpoint <- 1000; refuse(survival_expr_compute(b),"OS_INVALID_DESIGN")
check(check_calls(list.files(file.path(mod,"scripts"),pattern="\\.R$",full.names=TRUE),file.path(mod,"scripts/calls_registry.json"))$ok,"call registry")
check(check_default_drift(file.path(mod,"scripts/calls_registry.json"))$ok,"native API defaults")
check(!file.exists("Rplots.pdf"),"no device leak")
cat("SMOKE PASSED: expression-handoff survival\n")
unlink(tmp,recursive=TRUE)
